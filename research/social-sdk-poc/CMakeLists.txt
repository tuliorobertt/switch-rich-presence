cmake_minimum_required(VERSION 3.20)
project(SwitchPresenceSocialProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(DISCORD_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/sdk/discord_social_sdk" CACHE PATH "Pasta do Discord Social SDK")
set(SDK_HEADER "${DISCORD_SDK_ROOT}/include/discordpp.h")
set(SDK_LIB "${DISCORD_SDK_ROOT}/lib/release/discord_partner_sdk.lib")
set(SDK_DLL "${DISCORD_SDK_ROOT}/bin/release/discord_partner_sdk.dll")
foreach(path IN ITEMS "${SDK_HEADER}" "${SDK_LIB}" "${SDK_DLL}")
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Arquivo do SDK nao encontrado: ${path}. Consulte o README do teste.")
    endif()
endforeach()
add_executable(SocialPresenceProbe main.cpp)
target_include_directories(SocialPresenceProbe PRIVATE "${DISCORD_SDK_ROOT}/include")
target_link_libraries(SocialPresenceProbe PRIVATE "${SDK_LIB}")
add_custom_command(TARGET SocialPresenceProbe POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SDK_DLL}" "$<TARGET_FILE_DIR:SocialPresenceProbe>")
